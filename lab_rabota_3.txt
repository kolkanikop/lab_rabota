package main

import (
    "log"
    "net/http"
)

func main() {
    // Раздаём файлы из текущей директории (где лежит index.html)
    fs := http.FileServer(http.Dir("."))
    
    // Регистрируем обработчик для всех путей
    http.Handle("/", fs)

    // Проверяем наличие сертификатов для HTTPS
    log.Println("Сервер запущен на https://localhost:1234")
    
    // Используем ListenAndServeTLS для HTTPS
    err := http.ListenAndServeTLS("localhost:1234", "localhost+2.pem", "localhost+2-key.pem", nil)
    if err != nil {
        log.Fatal("Ошибка запуска сервера: ", err)
    }
}
